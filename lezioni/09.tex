\begin{frame}[fragile]
	\frametitle{Linguaggi e Computabilità: Lezione 09}
\begin{enumerate}
\item
Da Espressioni regolari a $\epsilon$-NFA
\item
Albero associato ad una espressione regolare
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Esercizio 1}
Costruire un $\epsilon$-NFA che riconosce il linguaggio dell'espressione regolare
\[
	\zero\uno\zero\kleene
\]
\end{frame}

\begin{frame}
\frametitle{Esercizio 2}
Costruire un $\epsilon$-NFA che riconosce il linguaggio dell'espressione regolare
\[
	\uno(\zero + \uno)\zero
\]
\end{frame}

\begin{frame}
\frametitle{Esercizio 3}
Costruire un $\epsilon$-NFA che riconosce il linguaggio dell'espressione regolare
\[
	\zero(\zero + \epsilon)(\uno + \epsilon)\kleene
\]
\end{frame}
